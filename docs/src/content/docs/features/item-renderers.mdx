Some widgets render a list of items — `dropdown`, `list`, `radiogroup`, and similar. By default, each item is rendered as a label/value pair. Item renderers let you supply a richer template (e.g. a country flag next to its label, a product card with a thumbnail and a price).

Item renderers live in `formConfig.itemRenderers`, keyed by a name you reference from the widget's `itemRenderer` prop.

## Wiring an item renderer

Build a `formConfig` that registers your renderer, reference its name from the dropdown's `itemRenderer` prop, and pass `formConfig` into your form component:

<IntegrationTabs>
  <Fragment slot="react">
    ```tsx
    import { gui } from '@golemui/gui-shared';
    import * as React from '@golemui/gui-react';
    import { CountryItemRenderer } from './CountryItemRenderer';

    const formDef = [
      gui.inputs.dropdown('country', {
        items: [
          { id: 'US', flag: '🇺🇸', label: 'United States' },
          { id: 'CA', flag: '🇨🇦', label: 'Canada' },
        ],
        labelField: 'label',
        valueField: 'id',
        itemRenderer: 'countryItemRenderer', // <- name registered below
      }),
    ];

    const formConfig = {
      widgetLoaders: React.widgetLoaders,
      itemRenderers: {
        countryItemRenderer: CountryItemRenderer, // React component
      },
    };

    export function MyForm() {
      return (
        <React.FormComponent
          formDef={formDef}
          data={{}}
          formConfig={formConfig}
        />
      );
    }
    ```
  </Fragment>
  <Fragment slot="angular">
    ```ts
    import { Component } from '@angular/core';
    import { CommonModule } from '@angular/common';
    import { gui } from '@golemui/gui-shared';
    import * as Angular from '@golemui/gui-angular';
    import { CountryItemRenderer } from './country-item-renderer.component';

    @Component({
      imports: [CommonModule, Angular.FormComponent],
      selector: 'app-my-form',
      template: `<gui-form
        [formDef]="formDef"
        [data]="formData"
        [formConfig]="formConfig"
      ></gui-form>`,
    })
    export class MyForm {
      protected formDef = [
        gui.inputs.dropdown('country', {
          items: [
            { id: 'US', flag: '🇺🇸', label: 'United States' },
            { id: 'CA', flag: '🇨🇦', label: 'Canada' },
          ],
          labelField: 'label',
          valueField: 'id',
          itemRenderer: 'countryItemRenderer',
        }),
      ];
      protected formData = {};
      protected formConfig = {
        widgetLoaders: Angular.widgetLoaders,
        itemRenderers: {
          countryItemRenderer: CountryItemRenderer, // Angular component class
        },
      };
    }
    ```
  </Fragment>
  <Fragment slot="lit">
    ```ts
    import { LitElement, html } from 'lit';
    import { customElement } from 'lit/decorators.js';
    import { gui } from '@golemui/gui-shared';
    import * as Lit from '@golemui/gui-lit';
    import { countryItemRenderer } from './country-item-renderer';

    @customElement('my-form')
    export class MyForm extends LitElement {
      formDef = [
        gui.inputs.dropdown('country', {
          items: [
            { id: 'US', flag: '🇺🇸', label: 'United States' },
            { id: 'CA', flag: '🇨🇦', label: 'Canada' },
          ],
          labelField: 'label',
          valueField: 'id',
          itemRenderer: 'countryItemRenderer',
        }),
      ];
      formData = {};
      formConfig = {
        widgetLoaders: Lit.widgetLoaders,
        itemRenderers: {
          countryItemRenderer, // Lit render function
        },
      };

      override createRenderRoot() {
        return this;
      }

      override render() {
        return html`
          <gui-form
            .formDef=${this.formDef}
            .data=${this.formData}
            .formConfig=${this.formConfig}
          ></gui-form>
        `;
      }
    }
    ```
  </Fragment>
